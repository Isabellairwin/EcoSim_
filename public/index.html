require("dotenv").config(); const express = require("express"); const mongoose = require("mongoose"); const http = require("http"); const { Server } = require("socket.io"); const app = express(); const server = http.createServer(app); const io = new Server(server); // ---------- DATABASE ---------- const Reading = mongoose.model("Reading", { traffic: Number, power: Number, water: Number, aqi: Number, scenario: String, time: { type: Date, default: Date.now }, }); mongoose.connect(process.env.MONGO_URI) .then(() => console.log("MongoDB connected")) .catch((err) => console.log("Mongo error:", err.message)); // ---------- DASHBOARD PAGE ---------- const PAGE = ` Ecosim Dashboard

Ecosim: Smart City Dashboard

Traffic %

-

Power %

-

Water %

-

AQI

-

Scenarios

Alerts

None
`; app.get("/", (req, res) => res.send(PAGE)); // ---------- SCENARIO CONTROL ---------- let scenario = "normal"; io.on("connection", (socket) => { socket.on("scenario", (s) => { scenario = s; console.log("Scenario:", s); }); }); // ---------- SIMULATION ENGINE (every 2 seconds) ---------- setInterval(async () => { const hour = new Date().getHours(); const rush = (hour >= 8 && hour <= 10) || (hour >= 17 && hour <= 19); const data = { traffic: Math.min(100, (rush ? 70 : 30) + Math.random() * 25), power: 40 + Math.random() * 40 + (hour >= 18 ? 15 : 0), water: 30 + Math.random() * 40, aqi: 40 + Math.random() * 80, }; // scenarios if (scenario === "rush") data.traffic = Math.min(100, data.traffic + 30); if (scenario === "outage") data.power = Math.min(100, data.power + 40); if (scenario === "pollution") data.aqi += 80; // smart rules const alerts = []; if (data.traffic > 80) alerts.push("Traffic high: extended green signal"); if (data.aqi > 100) alerts.push("Air quality poor: pollution alert"); if (data.power > 85) alerts.push("Power peak: rerouting load"); try { await Reading.create({ ...data, scenario }); } catch (e) { console.log("Save error:", e.message); } io.emit("reading", { ...data, alerts, scenario }); console.log( "Saved:", data.traffic.toFixed(1), "| Scenario:", scenario, "| Alerts:", alerts.length ? alerts.join(", ") : "none" ); }, 2000); server.listen(3000, () => console.log("Running on http://localhost:3000"));
